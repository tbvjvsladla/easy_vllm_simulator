"""Controlled stdlib IO primitive 1: the raw writes a text file makes for chunk sequences (success only). No owner logic, no json/yaml."""
import io, json, sys

class Raw(io.RawIOBase):
    def __init__(self):
        self.calls = []
    def writable(self):
        return True
    def write(self, data):
        self.calls.append(len(data))
        return len(data)

def run(name, bufsize, chunks, flush_after=()):
    raw = Raw()
    buf = io.BufferedWriter(raw, buffer_size=bufsize)
    text = io.TextIOWrapper(buf, encoding='utf-8')
    steps = []
    for index, chunk in enumerate(chunks):
        text.write(chunk)
        if index in flush_after:
            text.flush()
        steps.append(list(raw.calls))
    before_close = list(raw.calls)
    text.close()
    return dict(name=name, buffer_size=bufsize, chunk_chars=[len(c) for c in chunks], chunk_bytes=[len(c.encode()) for c in chunks],
                flush_after=list(flush_after), raw_after_each_chunk=steps, raw_before_close=before_close, raw_after_close=list(raw.calls))

scenarios = []
for bufsize in (4096, 8192):
    scenarios.append(run('ten-byte chunks x1000', bufsize, ['x' * 10] * 1000))
    scenarios.append(run('5000,5000,10', bufsize, ['a' * 5000, 'b' * 5000, 'c' * 10]))
    scenarios.append(run('100 then 9000', bufsize, ['a' * 100, 'b' * 9000]))
    scenarios.append(run('100 then 20000', bufsize, ['a' * 100, 'b' * 20000]))
    scenarios.append(run('3000 two-byte chars x2', bufsize, ['é' * 3000, 'é' * 3000]))
    scenarios.append(run('3000 flush 3000', bufsize, ['a' * 3000, 'b' * 3000], flush_after=(0,)))
for sizes in ([8191, 1, 1], [8192], [8193], [4095, 2], [4096], [4097], [1, 8192], [8000, 100, 100], [9000, 1, 9000]):
    scenarios.append(run('edges %s' % sizes, 4096, ['z' * n for n in sizes]))
print(json.dumps(dict(module=io.TextIOWrapper.__module__, chunk_size=io.TextIOWrapper(io.BytesIO())._CHUNK_SIZE, scenarios=scenarios)))
